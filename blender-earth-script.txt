import bpy
import math

# ============================================
# TEXTURE PATHS
# ============================================
earth_texture_path = r"INSERT YOUR OWN HERE"
space_texture_path = r"INSERT YOUR OWN HERE"

# ============================================
# CLEAN SCENE
# ============================================
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# ============================================
# CREATE EARTH
# ============================================
bpy.ops.mesh.primitive_uv_sphere_add(radius=1, location=(0, 0, 0))
earth = bpy.context.active_object
earth.name = "Earth"

bpy.ops.object.shade_smooth()

# ============================================
# EARTH MATERIAL
# ============================================
mat = bpy.data.materials.new(name="EarthMaterial")
mat.use_nodes = True

nodes = mat.node_tree.nodes
links = mat.node_tree.links
nodes.clear()

output = nodes.new(type='ShaderNodeOutputMaterial')
bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
tex = nodes.new(type='ShaderNodeTexImage')

tex.image = bpy.data.images.load(earth_texture_path)

links.new(tex.outputs['Color'], bsdf.inputs['Base Color'])
links.new(bsdf.outputs['BSDF'], output.inputs['Surface'])

earth.data.materials.append(mat)

# ============================================
# ANIMATION SETUP
# ============================================
bpy.context.scene.frame_start = 1
bpy.context.scene.frame_end = 365

# Clear rotation
earth.rotation_euler = (0, 0, 0)
earth.keyframe_insert(data_path="rotation_euler", frame=1)

# Full rotation (1 year simulation)
earth.rotation_euler = (0, 0, math.radians(360))
earth.keyframe_insert(data_path="rotation_euler", frame=365)

# ============================================
# SAFE FCURVE HANDLING (BLENDER 5+)
# ============================================
if earth.animation_data is None:
    earth.animation_data_create()

action = earth.animation_data.action

if action:
    # Ensure Z rotation fcurve exists safely
    fcurve = action.fcurve_ensure_for_datablock(
        datablock=earth,
        data_path="rotation_euler",
        index=2
    )

    # Make motion constant speed (no easing)
    for kp in fcurve.keyframe_points:
        kp.interpolation = 'LINEAR'

    # Make it loop forever (Cycles modifier)
    fcurve.modifiers.new(type='CYCLES')

# ============================================
# WORLD (SPACE BACKGROUND)
# ============================================
world = bpy.context.scene.world
if world is None:
    world = bpy.data.worlds.new("World")
    bpy.context.scene.world = world

world.use_nodes = True

wnodes = world.node_tree.nodes
wlinks = world.node_tree.links
wnodes.clear()

w_output = wnodes.new(type='ShaderNodeOutputWorld')
w_bg = wnodes.new(type='ShaderNodeBackground')
w_tex = wnodes.new(type='ShaderNodeTexEnvironment')

w_tex.image = bpy.data.images.load(space_texture_path)

wlinks.new(w_tex.outputs['Color'], w_bg.inputs['Color'])
wlinks.new(w_bg.outputs['Background'], w_output.inputs['Surface'])

# ============================================
# LIGHT
# ============================================
bpy.ops.object.light_add(type='SUN', location=(5, 5, 5))
