def mat_tex(name, path, emission=False, alpha=False):

    mat = bpy.data.materials.new(name)
    mat.use_nodes = True

    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()

    tex = nodes.new("ShaderNodeTexImage")
    tex.image = bpy.data.images.load(path)
    tex.image.colorspace_settings.name = 'sRGB'

    output = nodes.new("ShaderNodeOutputMaterial")

    if emission:
        em = nodes.new("ShaderNodeEmission")
        em.inputs["Strength"].default_value = 20
        links.new(tex.outputs["Color"], em.inputs["Color"])
        links.new(em.outputs["Emission"], output.inputs["Surface"])
    else:
        bsdf = nodes.new("ShaderNodeBsdfPrincipled")
        bsdf.inputs["Roughness"].default_value = 1.0

        links.new(tex.outputs["Color"], bsdf.inputs["Base Color"])

        if alpha:
            links.new(tex.outputs["Alpha"], bsdf.inputs["Alpha"])
            mat.surface_render_method = 'DITHERED'

        links.new(bsdf.outputs["BSDF"], output.inputs["Surface"])

    return mat

def planet(name, size, dist, tex):

    bpy.ops.mesh.primitive_uv_sphere_add(radius=size, location=(dist, 0, 0))
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(mat_tex(name, tex))
    obj.parent = root

    bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0, 0, 0))
    orbit = bpy.context.object
    orbit.name = name + "_Orbit"
    orbit.parent = root

    obj.parent = orbit

    return obj, orbit

def make_orbit_ring(name, dist, color, segments=128):
    curve_data = bpy.data.curves.new(name + "_RingCurve", type='CURVE')
    curve_data.dimensions = '3D'
    curve_data.resolution_u = 12
    curve_data.bevel_depth = 0.025
    curve_data.fill_mode = 'FULL'

    spline = curve_data.splines.new('POLY')
    spline.points.add(segments - 1)

    for i in range(segments):
        angle = 2 * math.pi * i / segments
        x = dist * math.cos(angle)
        y = dist * math.sin(angle)
        spline.points[i].co = (x, y, 0, 1)

    spline.use_cyclic_u = True

    ring_obj = bpy.data.objects.new(name + "_RingPath", curve_data)
    bpy.context.collection.objects.link(ring_obj)

    ring_mat = bpy.data.materials.new(name + "_RingPathMaterial")
    ring_mat.use_nodes = True
    rn = ring_mat.node_tree.nodes
    rl = ring_mat.node_tree.links
    rn.clear()
    emission = rn.new("ShaderNodeEmission")
    output = rn.new("ShaderNodeOutputMaterial")
    emission.inputs["Color"].default_value = color
    emission.inputs["Strength"].default_value = 1.0
    rl.new(emission.outputs["Emission"], output.inputs["Surface"])

    # Also set the material's viewport display color so the ring shows
    # bright yellow even under plain Solid shading, which ignores node
    # materials entirely and only reads diffuse_color.
    ring_mat.diffuse_color = color

    ring_obj.data.materials.append(ring_mat)

    ring_obj.parent = orbit_rings_group

    # Visible in both viewport and final render
    ring_obj.hide_render = False
    ring_obj.hide_select = True
    ring_obj.color = color
    ring_obj.display_type = 'SOLID'
    ring_obj.show_wire = True
    ring_obj.show_in_front = True

    return ring_obj

# -------------------------
# BUILD ONE METEOR (ROCK + TRAIL)
# -------------------------
def make_meteor(index, start, end, frame_in, frame_out):

    bpy.ops.mesh.primitive_ico_sphere_add(
        radius=random.uniform(0.08, 0.22),
        subdivisions=1,
        location=start
    )
    rock = bpy.context.object
    rock.name = f"Meteor_{index:02d}_Rock"
    rock.data.materials.append(meteor_mat)
    rock.parent = meteor_group

    # Random lumpy look: nudge each vertex slightly
    for v in rock.data.vertices:
        jitter = random.uniform(0.85, 1.15)
        v.co.x *= jitter
        v.co.y *= random.uniform(0.85, 1.15)
        v.co.z *= random.uniform(0.85, 1.15)

    # Glowing trail: a thin cone pointing back along the flight direction
    direction = (
        start[0] - end[0],
        start[1] - end[1],
        start[2] - end[2],
    )
    length = math.sqrt(sum(c * c for c in direction)) or 1.0
    unit = tuple(c / length for c in direction)

    bpy.ops.mesh.primitive_cone_add(
        radius1=0.0,
        radius2=rock.dimensions.x * 0.5 if rock.dimensions.x else 0.1,
        depth=1.4,
        location=start
    )
    trail = bpy.context.object
    trail.name = f"Meteor_{index:02d}_Trail"
    trail.data.materials.append(trail_mat)
    trail.parent = meteor_group

    # Point the cone's axis (local +Z) along the direction of travel
    trail.rotation_mode = 'QUATERNION'
    z_axis = (0, 0, 1)
    cross = (
        z_axis[1] * unit[2] - z_axis[2] * unit[1],
        z_axis[2] * unit[0] - z_axis[0] * unit[2],
        z_axis[0] * unit[1] - z_axis[1] * unit[0],
    )
    dot = z_axis[0] * unit[0] + z_axis[1] * unit[1] + z_axis[2] * unit[2]
    cross_len = math.sqrt(sum(c * c for c in cross))
    if cross_len > 1e-6:
        angle = math.atan2(cross_len, dot)
        axis_n = tuple(c / cross_len for c in cross)
        s = math.sin(angle / 2)
        trail.rotation_quaternion = (math.cos(angle / 2), axis_n[0]*s, axis_n[1]*s, axis_n[2]*s)

    # Offset the trail half its length behind the rock so it streaks back
    trail.location = (
        start[0] + unit[0] * 0.7,
        start[1] + unit[1] * 0.7,
        start[2] + unit[2] * 0.7,
    )

    # -------------------------
    # ANIMATE: streak from start to end, then disappear, then loop
    # -------------------------
    for obj, offset in ((rock, (0, 0, 0)), (trail, (unit[0]*0.7, unit[1]*0.7, unit[2]*0.7))):

        obj.location = (
            start[0] + offset[0],
            start[1] + offset[1],
            start[2] + offset[2],
        )
        obj.keyframe_insert(data_path="location", frame=frame_in)

        obj.location = (
            end[0] + offset[0],
            end[1] + offset[1],
            end[2] + offset[2],
        )
        obj.keyframe_insert(data_path="location", frame=frame_out)

        # Hide before it appears and after it finishes, so it doesn't
        # just sit visible at start/end position outside its window
        obj.scale = (1, 1, 1)
        obj.keyframe_insert(data_path="scale", frame=frame_in)
        obj.scale = (1, 1, 1)
        obj.keyframe_insert(data_path="scale", frame=frame_out - 1)
        obj.scale = (0, 0, 0)
        obj.keyframe_insert(data_path="scale", frame=frame_out)
        obj.scale = (0, 0, 0)
        obj.keyframe_insert(data_path="scale", frame=max(frame_in - 1, 1))

    return rock, trail

# GENERATE A SHOWER OF METEORS

num_meteors = 150

for i in range(num_meteors):

    # Random entry point from far outside the system, random exit
    # point passing roughly through the inner solar system
    theta = random.uniform(0, 2 * math.pi)
    entry_radius = random.uniform(45, 60)
    exit_radius = random.uniform(5, 20)

    start = (
        entry_radius * math.cos(theta),
        entry_radius * math.sin(theta),
        random.uniform(8, 18),
    )
    end = (
        exit_radius * math.cos(theta + random.uniform(-0.6, 0.6)),
        exit_radius * math.sin(theta + random.uniform(-0.6, 0.6)),
        random.uniform(-4, 4),
    )

    streak_duration = random.randint(15, 35)
    frame_in = random.randint(1, max(shower_span - streak_duration - 1, 1))
    frame_out = frame_in + streak_duration

    make_meteor(i, start, end, frame_in, frame_out)
