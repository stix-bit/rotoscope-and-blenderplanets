Blender Python Script

Below is the Python script I used in Blender to create and animate the Solar System. The script includes texture paths, material creation, planet modeling, orbit setup, and animation keyframes for both orbital and rotational movement. 


      import bpy
      import math

      # =========================================================
      # 🔧 TEXTURE PATHS
      # =========================================================
      T = {
        "sun": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_sun.jpg",
        "mercury": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_mercury.jpg",
        "venus": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_venus_surface.jpg",
        "earth": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_earth_daymap.jpg",
        "mars": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_mars.jpg",
        "jupiter": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_jupiter.jpg",
        "saturn": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_saturn.jpg",
        "uranus": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_uranus.jpg",
        "neptune": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_neptune.jpg",
        "pluto": r"C:\Users\johns\OneDrive\Documents\blender planet textures\plutomap1k.jpg",
        "moon": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_moon.jpg",
        "rings": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_saturn_ring_alpha.png",
        "space": r"C:\Users\johns\OneDrive\Documents\blender planet textures\2k_stars_milky_way.jpg"
      }

      # =========================================================
      # CLEAN SCENE
      # =========================================================
      bpy.ops.object.select_all(action='SELECT')
      bpy.ops.object.delete(use_global=False)

      # =========================================================
      # MATERIAL CREATION
      # =========================================================
      def mat_tex(name, path, emission=False):
        mat = bpy.data.materials.new(name)
        mat.use_nodes = True

        nodes = mat.node_tree.nodes
        links = mat.node_tree.links
        nodes.clear()

        tex = nodes.new("ShaderNodeTexImage")
        tex.image = bpy.data.images.load(path)
        tex.image.colorspace_settings.name = 'sRGB'

        output = nodes.new("ShaderNodeOutputMaterial")

        if emission:
          em = nodes.new("ShaderNodeEmission")

          links.new(tex.outputs["Color"], em.inputs["Color"])
          links.new(em.outputs["Emission"], output.inputs["Surface"])

        else:
          bsdf = nodes.new("ShaderNodeBsdfPrincipled")

          bsdf.inputs["Roughness"].default_value = 1.0

          # SAFE specular handling (no crashes)
          if "Specular IOR Level" in bsdf.inputs:
            bsdf.inputs["Specular IOR Level"].default_value = 0.5

          links.new(tex.outputs["Color"], bsdf.inputs["Base Color"])
          links.new(bsdf.outputs["BSDF"], output.inputs["Surface"])

        return mat

      # =========================================================
      # PLANET CREATION
      # =========================================================
      def planet(name, size, dist, tex):
        bpy.ops.mesh.primitive_uv_sphere_add(radius=size, location=(dist, 0, 0))
        obj = bpy.context.object
        obj.name = name
        obj.data.materials.append(mat_tex(name+"_mat", tex))
        return obj

      def orbit(obj):
        bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0,0,0))
        o = bpy.context.object
        obj.parent = o
        return o

      # =========================================================
      # SPACE BACKGROUND
      # =========================================================
      world = bpy.context.scene.world
      world.use_nodes = True

      nodes = world.node_tree.nodes
      links = world.node_tree.links
      nodes.clear()

      bg = nodes.new("ShaderNodeBackground")
      env = nodes.new("ShaderNodeTexEnvironment")
      out = nodes.new("ShaderNodeOutputWorld")

      env.image = bpy.data.images.load(T["space"])
      env.image.colorspace_settings.name = 'sRGB'

      bg.inputs["Strength"].default_value = 1.5

      links.new(env.outputs["Color"], bg.inputs["Color"])
      links.new(bg.outputs["Background"], out.inputs["Surface"])

      # =========================================================
      # SUN
      # =========================================================
      sun = planet("Sun", 3, 0, T["sun"])
      sun.data.materials[0] = mat_tex("sun_mat", T["sun"], emission=True)

      bpy.ops.object.light_add(type='SUN', location=(0,0,0))
      light = bpy.context.object

      light.data.energy = 3
      light.rotation_euler = (math.radians(45), math.radians(30), 0)

      # =========================================================
      # PLANETS (REAL ORBIT RATIOS)
      # orbital_period = Earth years (scaled)
      # rotation_period = hours (sign = direction)
      # =========================================================
      data = [
        ("Mercury", 0.3, 5, 0.24, 1407),
        ("Venus",   0.5, 7, 0.62, -5832),
        ("Earth",   0.5, 9, 1.0, 24),
        ("Mars",    0.4, 11, 1.88, 24.6),
        ("Jupiter", 1.5, 15, 11.86, 9.9),
        ("Saturn",  1.2, 19, 29.46, 10.7),
        ("Uranus",  1.0, 23, 84.01, -17.2),
        ("Neptune", 1.0, 27, 164.8, 16.1),
        ("Pluto",   0.2, 31, 248.0, -153),
      ]

      planets = {}

      for n, size, dist, orbit_p, rot_p in data:
        obj = planet(n, size, dist, T[n.lower()])
        o = orbit(obj)
        planets[n] = (obj, o, orbit_p, rot_p)

      # =========================================================
      # EARTH MOON
      # =========================================================
      earth = planets["Earth"][0]

      bpy.ops.mesh.primitive_uv_sphere_add(radius=0.15)
      moon = bpy.context.object
      moon.name = "Moon"
      moon.data.materials.append(mat_tex("moon_mat", T["moon"]))

      moon.parent = earth
      moon.location = (1.5, 0, 0)  # local orbit around Earth

      # =========================================================
      # SATURN RINGS
      # =========================================================
      import math

      saturn = planets["Saturn"][0]

      bpy.ops.object.empty_add(type='PLAIN_AXES', location=saturn.location)
      ring_anchor = bpy.context.object
      ring_anchor.name = "Saturn_Ring_Anchor"

      # IMPORTANT: follow Saturn orbit movement
      ring_anchor.parent = saturn.parent

      bpy.ops.mesh.primitive_torus_add(
        major_radius=2.2,
        minor_radius=0.05,
        location=saturn.location
      )

      rings = bpy.context.object
      rings.name = "Saturn_Rings"

      rings.data.materials.append(mat_tex("ring_mat", T["rings"]))

      rings.parent = ring_anchor

      # reset local position
      rings.location = (0, 0, 0)

      # flat ring (XY plane)
      rings.rotation_euler = (math.radians(90), 0, 0)

      # Saturn axial tilt (safe way)
      tilt = math.radians(26.7)
      rings.rotation_euler = (math.radians(90 + 26.7), 0, 0)

      rings.scale = (1.5, 1.5, 1.5)

      # =========================================================
      # ANIMATION SETTINGS
      # =========================================================
      frame_end = 360
      bpy.context.scene.frame_end = frame_end

      for name, (obj, orb, orbit_p, rot_p) in planets.items():

        # ORBIT SPEED (scaled realistic ratios)
        orb.rotation_euler = (0,0,0)
        orb.keyframe_insert("rotation_euler", frame=1)

        orbit_speed = 360 / orbit_p
        orb.rotation_euler = (0,0,math.radians(orbit_speed))
        orb.keyframe_insert("rotation_euler", frame=frame_end)

        # ROTATION
        obj.rotation_euler = (0,0,0)
        obj.keyframe_insert("rotation_euler", frame=1)

        direction = -1 if rot_p < 0 else 1
        spin_speed = 360 / abs(rot_p)

        obj.rotation_euler = (0,0,math.radians(spin_speed * frame_end * direction))
        obj.keyframe_insert("rotation_euler", frame=frame_end)

      # =========================================================
      # INFINITE LOOPING
      # =========================================================
      for obj in bpy.data.objects:
        ad = obj.animation_data
        if not ad:
          continue

        action = ad.action
        if not action or not hasattr(action, "fcurves"):
          continue

        for fc in action.fcurves:

          # ensure linear interpolation
          for kp in fc.keyframe_points:
            kp.interpolation = 'LINEAR'

          # remove old modifiers safely
          for m in list(fc.modifiers):
            fc.modifiers.remove(m)

          # add cycles
          mod = fc.modifiers.new(type='CYCLES')
          mod.mode_before = 'REPEAT'
          mod.mode_after = 'REPEAT'

      # =========================================================
      # CAMERA
      # =========================================================
      bpy.ops.object.camera_add(location=(0, -70, 25), rotation=(math.radians(70), 0, 0))
      bpy.context.scene.camera = bpy.context.object
      bpy.context.view_layer.update()
